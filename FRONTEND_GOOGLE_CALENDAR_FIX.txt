PROBLEM SUMMARY
================================================================================
The Google Calendar integration is failing because:

1. COOP SECURITY ISSUE: Browser Cross-Origin-Opener-Policy blocks window.closed 
   Error: "Cross-Origin-Opener-Policy policy would block the window.closed call"
   
2. INCOMPLETE OAUTH FLOW: After user authorizes on Google, the popup doesn't 
   properly communicate back to the main window
   
3. NO CALLBACK HANDLER: Frontend is missing a component to handle Google's OAuth 
   redirect and complete the authorization flow

4. BACKEND IS READY: The backend now allows the callback endpoint without JWT
   (/api/itineraries/calendar/oauth/callback is in permitAll)


WHAT HAPPENS NOW (BROKEN)
================================================================================
1. User clicks "Connect to Google Calendar" button
2. Frontend opens Google OAuth URL in popup
3. User authorizes on Google login page
4. Google redirects popup to: http://localhost:8081/api/itineraries/calendar/oauth/callback?code=XXX&state=1
5. BROKEN: No frontend callback page to handle this redirect
6. BROKEN: Popup window.closed check fails due to COOP policy
7. ERROR: "Error connecting to Google Calendar {message: 'Authorization cancelled by user.'}"


WHAT SHOULD HAPPEN (FIXED)
================================================================================
1. User clicks "Connect to Google Calendar" button
2. Frontend opens Google OAuth URL in popup using postMessage approach
3. User authorizes on Google login page
4. Google redirects popup to OAuth callback component
5. FIXED: Frontend callback component receives the code
6. FIXED: Callback component exchanges code for tokens via backend
7. FIXED: Callback component sends postMessage to main window with result
8. Main window receives message and closes popup
9. SUCCESS: "Connected to Google Calendar" message appears

STEP 1: CREATE NEW COMPONENT FOR OAuth CALLBACK
FILE: src/app/components/oauth-callback/oauth-callback.component.ts
CREATE THIS NEW FILE:

---BEGIN FILE---
import { Component, OnInit } from '@angular/core';
import { GoogleCalendarService } from 'path-to-your-service/google-calendar.service';

@Component({
  selector: 'app-oauth-callback',
  template: '<div style="text-align: center; margin-top: 50px;"><p>Processing authorization...</p></div>',
  styleUrls: []
})
export class OAuthCallbackComponent implements OnInit {
  constructor(private googleCalendarService: GoogleCalendarService) {}

  ngOnInit(): void {
    // Extract query parameters from URL
    const params = new URLSearchParams(window.location.search);
    const code = params.get('code');
    const state = params.get('state');
    const error = params.get('error');

    console.log('OAuth callback received:', { code: !!code, state, error });

    // If error from Google
    if (error) {
      console.error('OAuth error from Google:', error);
      if (window.opener) {
        window.opener.postMessage({
          type: 'GOOGLE_CALENDAR_AUTH_ERROR',
          error: error || 'Authorization cancelled by user'
        }, window.location.origin);
      }
      window.close();
      return;
    }

    // If we have code and state, exchange code for tokens
    if (code && state) {
      console.log('Exchanging authorization code for tokens...');
      
      this.googleCalendarService.completeOAuthCallback(code, state)
        .subscribe({
          next: (response) => {
            console.log('OAuth callback successful:', response);
            // Send success message to opener window
            if (window.opener) {
              window.opener.postMessage({
                type: 'GOOGLE_CALENDAR_AUTH_SUCCESS',
                payload: response
              }, window.location.origin);
            }
            // Close popup after brief delay
            setTimeout(() => window.close(), 500);
          },
          error: (err) => {
            console.error('Error exchanging authorization code:', err);
            // Send error message to opener window
            if (window.opener) {
              window.opener.postMessage({
                type: 'GOOGLE_CALENDAR_AUTH_ERROR',
                error: err.message || 'Failed to complete authorization'
              }, window.location.origin);
            }
            setTimeout(() => window.close(), 500);
          }
        });
    } else {
      // Missing code or state
      console.error('Missing authorization code or state parameter');
      if (window.opener) {
        window.opener.postMessage({
          type: 'GOOGLE_CALENDAR_AUTH_ERROR',
          error: 'Missing authorization code. Please try again.'
        }, window.location.origin);
      }
      window.close();
    }
  }
}

---BEGIN REPLACEMENT---
connectToGoogleCalendar(): Promise<any> {
  return new Promise((resolve, reject) => {
    // Get authorization URL from backend
    this.http.get<string>('/api/itineraries/calendar/auth-url')
      .subscribe({
        next: (authUrl: string) => {
          console.log('Opening Google Calendar authorization URL');
          
          // Calculate popup position to center it on screen
          const width = 500;
          const height = 600;
          const left = window.screenX + (window.outerWidth - width) / 2;
          const top = window.screenY + (window.outerHeight - height) / 2;
          
          // Open popup window
          const popup = window.open(
            authUrl,
            'GoogleCalendarAuth',
            `width=${width},height=${height},left=${left},top=${top}`
          );
          
          if (!popup) {
            reject(new Error('Failed to open authorization popup. Please check your browser popup blocker settings.'));
            return;
          }

          // Set up message listener to receive response from popup
          const messageHandler = (event: MessageEvent) => {
            // Security check: verify message comes from same origin
            if (!event.origin.includes(window.location.hostname)) {
              console.warn('Ignoring message from untrusted origin:', event.origin);
              return;
            }

            // Handle successful authorization
            if (event.data.type === 'GOOGLE_CALENDAR_AUTH_SUCCESS') {
              console.log('OAuth authorization successful');
              window.removeEventListener('message', messageHandler);
              clearTimeout(timeout);
              resolve(event.data.payload);
            } 
            // Handle authorization error
            else if (event.data.type === 'GOOGLE_CALENDAR_AUTH_ERROR') {
              console.error('OAuth authorization error:', event.data.error);
              window.removeEventListener('message', messageHandler);
              clearTimeout(timeout);
              reject(new Error(event.data.error || 'Authorization failed'));
            }
          };

          // Register message event listener
          window.addEventListener('message', messageHandler);

          // Timeout after 60 seconds (user takes too long to authorize)
          const timeout = setTimeout(() => {
            window.removeEventListener('message', messageHandler);
            reject(new Error('Authorization timeout. Please try again.'));
          }, 60000);
        },
        error: (err) => {
          reject(new Error('Failed to get authorization URL: ' + err.message));
        }
      });
  });
}
---END REPLACEMENT---


STEP 4: ADD NEW METHOD TO google-calendar.service.ts
FILE: src/app/services/google-calendar.service.ts

ADD THIS NEW METHOD to the GoogleCalendarService class:

---BEGIN NEW METHOD---
completeOAuthCallback(code: string, state: string): Observable<any> {
  console.log('Calling backend to exchange authorization code');
  return this.http.get<any>(
    '/api/itineraries/calendar/oauth/callback',
    { 
      params: { 
        code: code, 
        state: state 
      } 
    }
  ).pipe(
    catchError((error) => {
      console.error('Error in OAuth callback:', error);
      return throwError(() => error);
    })
  );
}
---END NEW METHOD---
